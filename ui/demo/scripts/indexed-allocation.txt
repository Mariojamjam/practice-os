# Difference from the course specification: every directory occupies a
# metadata block (the specification shows one only for '/'), so /system
# and /system/logs take blocks 01 and 02, and the file's inode and data
# blocks start two blocks later.
mkdir /system
mkdir /system/logs
create /system/logs/app.log --size 10KB
fs_status --details
